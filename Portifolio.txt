PORTIFÓLIO 3



programa
{
    funcao inicio()
    {
        cadeia nomes[3]
        real notas[3][2]
        real media
        inteiro i, j

        para (i = 0; i < 3; i++)
        {
            leia(nomes[i])

            para (j = 0; j < 2; j++)
            {
                leia(notas[i][j])
            }
        }

        para (i = 0; i < 3; i++)
        {
            media = (notas[i][0] + notas[i][1]) / 2
            escreva(nomes[i], " - Média: ", media, "\n")
        }
    }
}

programa
